/* ============================================================ floating-navigation.js — الزر العائم الذكي متكيف الألوان ============================================================ */ (function () { 'use strict'; // 1. فحص الصلاحيات من localStorage function checkMultiAccess() { try { var allowed = JSON.parse(localStorage.getItem('allowedPages') || '[]'); var portalSession = JSON.parse(localStorage.getItem('portalSession') || 'null'); if ((Array.isArray(allowed) && allowed.length > 1) || (portalSession && portalSession.role === 'admin')) { return true; } } catch (e) {} return false; } if (!checkMultiAccess()) return; // 2. استخراج لون هوية الموقع ديناميكياً function detectSiteBrandColor() { var metaTheme = document.querySelector('meta[name="theme-color"]'); if (metaTheme && metaTheme.getAttribute('content')) { return metaTheme.getAttribute('content'); } var header = document.querySelector('header, nav, .bg-primary, [class*="bg-"]'); if (header) { var bg = window.getComputedStyle(header).backgroundColor; if (bg && bg !== 'rgba(0, 0, 0, 0)' && bg !== 'transparent') { return bg; } } var primaryElem = document.querySelector('h1, a, button'); if (primaryElem) { var color = window.getComputedStyle(primaryElem).color; if (color && color !== 'rgba(0, 0, 0, 0)') { return color; } } return '#1e293b'; } // 3. إنشاء عناصر الزر العائم في DOM function initFloatingButton() { if (document.getElementById('smart-floating-nav')) return; var brandColor = detectSiteBrandColor(); var navContainer = document.createElement('div'); navContainer.id = 'smart-floating-nav'; navContainer.style.cssText = ` position: fixed; bottom: 20px; right: 20px; z-index: 999999; display: flex; align-items: center; background: ${brandColor}; color: #ffffff; border-radius: 50px; padding: 6px; box-shadow: 0 8px 24px rgba(0,0,0,0.35); font-family: system-ui, -apple-system, sans-serif; cursor: grab; user-select: none; touch-action: none; border: 1px solid rgba(255, 255, 255, 0.2); `; var iconCircle = document.createElement('div'); iconCircle.innerHTML = '🏠'; iconCircle.style.cssText = ` width: 40px; height: 40px; background: rgba(255, 255, 255, 0.2); border-radius: 50%; display: flex; align-items: center; justify-content: center; font-size: 20px; flex-shrink: 0; backdrop-filter: blur(4px); `; var expandedContent = document.createElement('div'); expandedContent.id = 'floating-nav-content'; expandedContent.style.cssText = ` display: none; align-items: center; padding: 0 14px 0 8px; white-space: nowrap; `; var labelText = document.createElement('span'); labelText.textContent = 'العودة للرئيسية ⬅️'; labelText.style.cssText = 'font-size: 14px; font-weight: 600; color: #ffffff;'; expandedContent.appendChild(labelText); navContainer.appendChild(iconCircle); navContainer.appendChild(expandedContent); document.body.appendChild(navContainer); var isExpanded = false; var collapseTimeout; function collapse() { isExpanded = false; expandedContent.style.display = 'none'; } function expand() { isExpanded = true; expandedContent.style.display = 'flex'; clearTimeout(collapseTimeout); collapseTimeout = setTimeout(collapse, 5000); } // ميزة السحب والتحريك المتزامنة فورياً (Draggable) var isDragging = false; var hasMoved = false; var startX, startY, initialLeft, initialTop; function onStart(e) { isDragging = true; hasMoved = false; var clientX = e.touches ? e.touches[0].clientX : e.clientX; var clientY = e.touches ? e.touches[0].clientY : e.clientY; var rect = navContainer.getBoundingClientRect(); startX = clientX; startY = clientY; initialLeft = rect.left; initialTop = rect.top; // إزالة أي transition أثناء السحب لضمان الاستجابة السريعة navContainer.style.transition = 'none'; navContainer.style.cursor = 'grabbing'; navContainer.style.right = 'auto'; navContainer.style.bottom = 'auto'; } function onMove(e) { if (!isDragging) return; var clientX = e.touches ? e.touches[0].clientX : e.clientX; var clientY = e.touches ? e.touches[0].clientY : e.clientY; var dx = clientX - startX; var dy = clientY - startY; if (Math.abs(dx) > 5 || Math.abs(dy) > 5) { hasMoved = true; } navContainer.style.left = (initialLeft + dx) + 'px'; navContainer.style.top = (initialTop + dy) + 'px'; } function onEnd() { if (!isDragging) return; isDragging = false; navContainer.style.cursor = 'grab'; } navContainer.addEventListener('mousedown', onStart); window.addEventListener('mousemove', onMove); window.addEventListener('mouseup', onEnd); navContainer.addEventListener('touchstart', onStart, { passive: true }); window.addEventListener('touchmove', onMove, { passive: true }); window.addEventListener('touchend', onEnd); // عند الضغط على الزر navContainer.addEventListener('click', function () { if (hasMoved) return; // عدم التنفيذ إذا كانت الإيماءة مجرد سحب if (!isExpanded) { // الضغطة الأولى: توسيع الزر expand(); } else { // الضغطة الثانية (عندما يكون متوسعاً): الانتقال المباشر للمنصة window.location.href = 'index.html'; } }); } if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', initFloatingButton); } else { initFloatingButton(); } })();
